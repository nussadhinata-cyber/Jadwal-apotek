    function applyFilter() {
      document.querySelectorAll(".shift-slot").forEach(slot => {

      });

        button.classList.toggle("bg-slate-800", selected);
        button.classList.toggle("text-white", selected);
        button.classList.toggle("border-slate-800", selected);
        button.classList.toggle("bg-white", !selected);
        button.classList.toggle("text-slate-600", !selected);
        button.classList.toggle("border-slate-300", !selected);
        button.setAttribute("aria-pressed", String(selected));
      });
    }

    function renderDashboard() {
      const year = visibleDate.getFullYear();
      const month = visibleDate.getMonth();
      const days = getMonthDays();
      let morningCount = 0;
      let afternoonCount = 0;
      const totals = Object.fromEntries(staff.map(name => [name, 0]));

      for (let day = 1; day <= days; day++) {
        const entry = assignments[dateKey(year, month, day)] || {};
        if (entry.morning) {
          morningCount++;
          totals[entry.morning] = (totals[entry.morning] || 0) + 1;
        }
        if (entry.afternoon) {
          afternoonCount++;
          totals[entry.afternoon] = (totals[entry.afternoon] || 0) + 1;
        }
      }

      const filled = morningCount + afternoonCount;
      document.getElementById("summary-days").textContent = days;
      document.getElementById("summary-morning").textContent = morningCount;
      document.getElementById("summary-afternoon").textContent = afternoonCount;
      document.getElementById("summary-filled").textContent = filled;
      document.getElementById("summary-empty").textContent = (days * 2) - filled;

      const list = document.getElementById("staff-summary-list");
      list.innerHTML = "";
      staff.forEach(name => {
        const item = document.createElement("div");
        item.className = "flex items-center justify-between rounded-lg bg-slate-50 px-3 py-2 text-sm";
        item.innerHTML = `<span class="font-semibold text-slate-700"></span><span class="rounded-full bg-white px-2 py-0.5 text-xs font-extrabold text-teal-700"></span>`;
        item.children[0].textContent = name;
        item.children[1].textContent = `${totals[name] || 0} shift`;
        list.appendChild(item);
      });
    }

    function saveStaffName(oldName, rawName) {
      const newName = rawName.trim().replace(/\s+/g, " ");
      if (!newName) {
        showToast("Nama pegawai tidak boleh kosong.");
        return;
      }
      if (staff.some(name => name !== oldName && name.toLowerCase() === newName.toLowerCase())) {
        showToast("Nama pegawai tersebut sudah ada.");
        return;
      }
      staff = staff.map(name => name === oldName ? newName : name);
      Object.values(assignments).forEach(entry => {
        if (entry.morning === oldName) entry.morning = newName;
        if (entry.afternoon === oldName) entry.afternoon = newName;
      });
      editingStaff = null;
      renderAll();
      showToast("Nama pegawai dan jadwal berhasil diperbarui.");
    }

    function deleteStaff(name) {
      staff = staff.filter(existing => existing !== name);
      Object.keys(assignments).forEach(key => {
        const entry = assignments[key];
        if (entry.morning === name) delete entry.morning;
        if (entry.afternoon === name) delete entry.afternoon;
        if (!entry.morning && !entry.afternoon) delete assignments[key];
      });
      pendingDeleteStaff = null;
      renderAll();
      showToast("Pegawai berhasil dihapus.");
    }

      event.preventDefault();
      const morning = document.getElementById("morning-time").value.trim();
      const afternoon = document.getElementById("afternoon-time").value.trim();

      if (!morning || !afternoon) {
        showToast("Jam Shift Pagi dan Shift Sore harus diisi.");
        return;
      }

      shiftTimes = { morning, afternoon };
      renderAll();
      showToast("Pengaturan jam shift berhasil disimpan.");
    });

    document.getElementById("staff-form").addEventListener("submit", event => {
      event.preventDefault();
      const input = document.getElementById("staff-name");
      const name = input.value.trim().replace(/\s+/g, " ");

      if (!name) {
        showToast("Masukkan nama petugas terlebih dahulu.");
        return;
      }
      if (staff.some(existing => existing.toLowerCase() === name.toLowerCase())) {
        showToast("Nama petugas tersebut sudah ada.");
        return;
      }

      staff.push(name);
      input.value = "";
      renderDashboard();
      renderAttendanceSummary();
      renderStaffSubtotals();
      renderStaffChips();
      showToast(`${name} ditambahkan ke daftar petugas.`);
    });

      event.preventDefault();
      if (!selectedShift) return;

      const name = document.getElementById("assignment-staff").value;
      if (!name) {
        showToast("Pilih petugas atau gunakan tombol hapus penugasan.");
        return;
      }

      if (!assignments[selectedShift.key]) assignments[selectedShift.key] = {};
      assignments[selectedShift.key][selectedShift.type] = name;
      closeAssignment();
      renderAll();
      showToast("Penugasan shift berhasil disimpan.");
    });

      const entry = assignments[selectedShift.key];
      if (entry) {
        delete entry[selectedShift.type];
        if (!entry.morning && !entry.afternoon) delete assignments[selectedShift.key];
      }
      closeAssignment();
      renderAll();
      showToast("Penugasan shift dihapus.");
    });
